2. 数据加密 (cloud 模式启用)

# 方案选择: 轻量级应用层加密
# 依赖: cryptography 包

# app/core/crypto.py
from cryptography.fernet import Fernet
import os

_fernet = None

def init_encryption(key: str = None):
    """初始化加密器。cloud 模式自动启用"""
    if key:
        _fernet = Fernet(key.encode())
    else:
        _fernet = None  # local 模式不加密

def encrypt_field(plain_text: str) -> str:
    if not _fernet: return plain_text
    return _fernet.encrypt(plain_text.encode()).decode()

def decrypt_field(cipher_text: str) -> str:
    if not _fernet: return cipher_text
    return _fernet.decrypt(cipher_text.encode()).decode()

# 适用范围: 日程 title、description 等敏感字段
# 索引字段（id, start_time, end_time, user_id）不加密以支持查询
# 加密对查询透明 — 应用层加解密，DB 存密文
