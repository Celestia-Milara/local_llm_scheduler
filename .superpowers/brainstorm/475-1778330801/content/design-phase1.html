1. @skill 装饰器 + 注册器

# skill/__init__.py
from typing import Callable, Dict, Any

_SKILL_REGISTRY: Dict[str, Dict[str, Any]] = {}

def skill(name: str, description: str, parameters: dict):
    """装饰器: 将函数注册为 Skill"""
    def decorator(func: Callable):
        _SKILL_REGISTRY[name] = {
            "function": func,
            "description": description,
            "parameters": parameters,
        }
        return func
    return decorator

def get_tools() -> list:
    """供 agent_engine 获取 LLM tool 定义"""
    return [
        {
            "type": "function",
            "function": {
                "name": name,
                "description": meta["description"],
                "parameters": meta["parameters"],
            }
        }
        for name, meta in _SKILL_REGISTRY.items()
    ]

def execute_tool(name: str, args: dict) -> str:
    return _SKILL_REGISTRY[name]["function"](**args)
